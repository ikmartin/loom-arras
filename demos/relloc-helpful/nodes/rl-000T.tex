\begin{definition}\label{rl-000T}\label{defn:euler-class-of-vector-bundle-stack}
  Let \(\mathcal X\) be an algebraic stack with a trivial \(T\)-action and let $V$ be a vector bundle stack over $\mathcal X$. We say that \(V\) is \emph{finite rank and moving} if, up to equivalence of vector bundle stacks, it can be represented as a quotient \([V_1/V_0]\) where \(V_0\) and \(V_1\) are each finite rank moving vector bundles over \(\mathcal X\). If \(V\) is such a vector bundle stack then we then define its Euler class to be
  \begin{align*}
    e_T(V) = e_T(V_1)/e_T(V_0) \in A^*_T(\mathcal X)_{\loc}.
  \end{align*}
\end{definition}
