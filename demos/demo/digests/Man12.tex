\section*{Overview}
Manolache constructs a virtual pullback for a morphism equipped with a perfect obstruction theory relative to the target, proves functoriality, and compares it with Gysin maps. This digest holds only the two results the demo cites.

\section{Preliminaries}\label{Man12-sec-2}

\begin{theorem}[{\cite[Standing assumptions, Section 2]{Man12}}]\label{Man12-setup}
Throughout, all stacks are of finite type over a field $k$, and every morphism is of Deligne--Mumford type.
\end{theorem}

\section{Virtual pullback}\label{Man12-sec-3}

\begin{proposition}[{\cite[Proposition 3.2, p.~9]{Man12}}]\label{Man12-prop-3.2}
\uses{Man12-setup}
Let $f\colon X\to Y$ be a closed embedding of Deligne--Mumford stacks. Then the fixed locus of any involution commuting with $f$ is closed in $X$.
\end{proposition}
