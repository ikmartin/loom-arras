\begin{definition}\label{rl-001C}
\cite{aranha-khan-latyntsev-etal_VirtualLocalizationRevisited2025}
  Let \(X\) be an Artin stack over \(S\) with a \(G\)-action. The \emph{(embedded) \(G\)-fixed locus} \(X^{eG}\subseteq X\) is the locus of \(X\) where the canonical homomorphism of group algebraic spaces over \(X\)
  \begin{align*}
    \St_{X,G}\to G \times_S X
  \end{align*}
  is surjective\footnote{that is, is an effective epimorphism of fppf sheaves.}.
\end{definition}
