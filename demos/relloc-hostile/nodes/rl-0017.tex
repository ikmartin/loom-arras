\begin{lem}[Obstruction theory for $\widetilde \pi:\widetilde M\to \mathcal Z)$]\label{rl-0017}\label{lem:obstruction-theory-for-tilde-M}
  Let \(\phi:\mathbb E\to L_{M/\mathcal Y}\) be a relative perfect obstruction theory for \(\pi:M\to \mathcal Y\). Then the composite map
  \begin{align*}
    \widetilde \phi:\widetilde{\mathbb E} := \tilde i^*\mathbb E \xrightarrow{\tilde i^*\phi} \widetilde i^*L_{M/\mathcal Y} \to L_{\widetilde M/\mathcal Z}
  \end{align*}
  is a perfect obstruction theory for \(\tilde \pi:\widetilde M\to \mathcal Z\)
\end{lem}
\begin{proof}
  \cite[Prop. 7.2]{behrend-fantechi_IntrinsicNormalCone1997}, \cite[Theorem 4.3]{manolache_VirtualPullbacks2012}

  \red{The BF proposition isn't technically stated for algebraic stacks but it probably works and is exactly what we want, the Manolache citation is the one that says virtual pullbacks are stable under base change which I think implicitly uses this pullback compatibility.}
\end{proof}
