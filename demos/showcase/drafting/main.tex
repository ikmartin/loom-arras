\title{Balanced flows on finite quivers}
\author{The loom showcase}

\begin{abstract}
We compute the rank of the lattice of balanced weightings of a finite quiver, show that it is a saturated sublattice of the weighting lattice, and count its bounded members. Nothing here is true by accident: the quiver is a showcase.
\end{abstract}

\section{Introduction}\label{sh-0100}

A quiver carries two lattices: the weightings of its arrows, and the balanced weightings, those that neither create nor destroy anything at a vertex. This paper is about the second. Our main result, Theorem~\ref{sh-0009}, computes its rank from the combinatorics of the underlying graph alone, and Theorem~\ref{sh-000C} counts the balanced weightings of bounded height.

The cycle lattice of a quiver is classical, and we follow \cite[Proposition 2.1]{Arden24} throughout; the counting argument of Section~\ref{sh-0300} is an adaptation of the median-order technique of \cite[Theorem 2.3]{Bellamy19}. We also use the exchange bound of \cite[Lemma 5.9]{Arden24}, which the digest of that paper does not carry. The survey \cite{Cortez07} lists what is known for infinite quivers, none of which we use.

\begin{convention}\label{sh-0001}
Every quiver in this paper is finite: both $V(\quiv)$ and $A(\quiv)$ are finite sets, and loops and parallel arrows are allowed. All lattices are free abelian groups of finite rank, written additively.
\end{convention}

\section{Weightings and the divergence map}\label{sh-0200}

\begin{definition}[Quiver]\label{sh-0002}
A \emph{quiver} $\quiv$ is a quadruple $(V, A, s, t)$ with $V$ and $A$ finite sets and $s, t \colon A \to V$ the source and target maps. A \emph{weighting} of $\quiv$ is a function $w \colon A \to \ZZ$, and the weightings form a lattice $W(\quiv) \cong \ZZ^{A}$ under pointwise addition.
\end{definition}

\begin{definition}[Divergence]\label{sh-0003}
The \emph{divergence} of a weighting $w$ at a vertex $v$ is
\begin{equation}\label{eq:divergence}
(\bd w)(v) \;=\; \sum_{a \colon t(a) = v} w(a) \;-\; \sum_{a \colon s(a) = v} w(a),
\end{equation}
the weight arriving at $v$ less the weight leaving it. The resulting homomorphism $\bd \colon W(\quiv) \to \ZZ^{V}$ is the \emph{divergence map}.
\end{definition}

\begin{definition}[Balanced weighting]\label{sh-0004}
A weighting $w$ is \emph{balanced} if $\bd w = 0$, and the balanced weightings form the sublattice
\begin{equation}\label{eq:cycle-lattice}
\Zcyc(\quiv) \;=\; \ker \bd \;\subseteq\; W(\quiv),
\end{equation}
which we call the \emph{cycle lattice} of $\quiv$.
\end{definition}

\begin{example}\label{sh-0005}
For the quiver with one vertex and $n$ loops, every weighting is balanced and $\Zcyc(\quiv) = \ZZ^{n}$. For a single arrow between two distinct vertices, only the zero weighting is balanced.
\end{example}

\begin{lemma}[Support]\label{sh-0006}
Let $w$ be a balanced weighting with $w \geq 0$ and $w \neq 0$. Then the support $\supp(w)$ contains a directed cycle of $\quiv$.
\end{lemma}
\begin{proof}
\uses{sh-0001}
Pick $a_{0}$ with $w(a_{0}) > 0$. Since $(\bd w)(t(a_{0})) = 0$ and the arrow $a_{0}$ contributes positively to that sum, some arrow $a_{1}$ with $s(a_{1}) = t(a_{0})$ has $w(a_{1}) > 0$. Repeating, we obtain an infinite walk inside $\supp(w)$, which is a finite set by Convention~\ref{sh-0001}, so some vertex repeats and the walk contains a directed cycle.
\end{proof}

\begin{proposition}[Saturation]\label{sh-0007}
$\Zcyc(\quiv)$ is a saturated sublattice of $W(\quiv)$: if $n w \in \Zcyc(\quiv)$ for some $w \in W(\quiv)$ and some integer $n \neq 0$, then $w \in \Zcyc(\quiv)$.
\end{proposition}
\begin{proof}[First proof]
The divergence map is a homomorphism into a torsion-free group, so $\bd(n w) = n \, \bd w = 0$ forces $\bd w = 0$.
\end{proof}
\begin{proof}[Second proof of Proposition~\ref{sh-0007}]\label{sh-0010}
\uses{sh-0004}
The quotient $W(\quiv) / \Zcyc(\quiv)$ embeds in $\ZZ^{V}$ by the first isomorphism theorem, hence is torsion-free, and a sublattice with torsion-free quotient is saturated. The cycle lattice is the kernel of Definition~\ref{sh-0004}, so nothing else is needed.
\end{proof}

\begin{remark}\label{sh-0008}
Saturation is what makes the rank in Theorem~\ref{sh-0009} an honest dimension count: over $\RR$ the same kernel has the same rank, and no index creeps in.
\end{remark}

\section{The rank of the cycle lattice}\label{sh-0300}

\begin{theorem}[Rank]\label{sh-0009}
Let $\quiv$ be a quiver whose underlying graph has $c$ connected components. Then
\begin{equation}\label{eq:rank}
\rank \Zcyc(\quiv) \;=\; |A(\quiv)| - |V(\quiv)| + c.
\end{equation}
\end{theorem}
\begin{proof}
\uses{sh-0007}
The image of $\bd$ is the set of integer vectors summing to zero on each component, a sublattice of $\ZZ^{V}$ of rank $|V(\quiv)| - c$; this is \cite[Proposition 2.1]{Arden24}. Since $\Zcyc(\quiv) = \ker \bd$ is saturated by Proposition~\ref{sh-0007}, the rank of the kernel is the rank of the source less the rank of the image, which gives \eqref{eq:rank}.
\end{proof}

\begin{corollary}[Trees]\label{sh-000A}
A quiver whose underlying graph is a forest has $\Zcyc(\quiv) = 0$.
\end{corollary}
\begin{proof}
\uses{sh-0009}
A forest on $|V(\quiv)|$ vertices with $c$ components has exactly $|V(\quiv)| - c$ edges, so Theorem~\ref{sh-0009} and \eqref{eq:rank} give rank zero, and a saturated sublattice of rank zero is trivial.
\end{proof}

\subsection{Bounded balanced weightings}\label{sh-0301}

\begin{definition}[Height]\label{sh-000B}
The \emph{height} of a weighting $w$ is $\|w\| = \max_{a \in A(\quiv)} |w(a)|$, and $N_{\quiv}(k)$ denotes the number of balanced weightings of height at most $k$.
\end{definition}

\begin{theorem}[Counting]\label{sh-000C}
For every quiver $\quiv$ there is a polynomial $P_{\quiv}$ of degree $\rank \Zcyc(\quiv)$ with $N_{\quiv}(k) = P_{\quiv}(k)$ for all $k \geq 0$.
\end{theorem}
\begin{proof}
\uses{sh-0009, sh-0007}
The balanced weightings of height at most $k$ are the lattice points of the dilate $k \Pi$ of the polytope $\Pi = \Zcyc(\quiv) \otimes \RR \cap [-1,1]^{A}$, which is a rational polytope of dimension $\rank \Zcyc(\quiv)$ by Theorem~\ref{sh-0009}. Ehrhart's theorem in the form of \cite[Theorem 3.2]{Bellamy19} applies, since $\Pi$ has integral vertices by the saturation of Proposition~\ref{sh-0007}.
\end{proof}

\begin{caveat}\label{sh-000D}
The polytope $\Pi$ need not have integral vertices when parallel arrows are weighted with different signs; the proof above quietly assumes it does, and Section~\ref{sh-0400} records what is needed.
\end{caveat}
