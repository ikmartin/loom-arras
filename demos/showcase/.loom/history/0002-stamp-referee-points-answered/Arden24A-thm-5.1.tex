\begin{theorem}[{\cite[Theorem 5.1, p.~3]{Arden24A}}]\label{Arden24A-thm-5.1}
\uses{Arden24A-lem-2.2, Arden24A-thm-3.1}
\label{Arden24A-thm:subquiver}
Let $P$ be a subquiver of a quiver $Q$ and $w$ a weighting of $P$. The extension of $w$ by zero is balanced on $Q$ if and only if $w$ is balanced on $P$. In particular the cycle space of $P$ is a saturated sublattice of the cycle space of $Q$, of rank $|A(P)| - |V(P)| + c(P)$.
\end{theorem}
