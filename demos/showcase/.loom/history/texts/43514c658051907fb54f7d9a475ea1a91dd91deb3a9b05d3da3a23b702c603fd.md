\begin{theorem}[Counting]\label{sh-000C}
For every quiver $\quiv$ there is a polynomial $P_{\quiv}$ of degree $\rank \Zcyc(\quiv)$ with $N_{\quiv}(k) = P_{\quiv}(k)$ for all $k \geq 0$.
\end{theorem}
\begin{proof}
\uses{sh-0009, sh-0007}
The balanced weightings of height at most $k$ are the lattice points of the dilate $k \Pi$ of the polytope $\Pi = \Zcyc(\quiv) \otimes \RR \cap [-1,1]^{A}$, which is a rational polytope of dimension $\rank \Zcyc(\quiv)$ by Theorem~\ref{sh-0009}. Ehrhart's theorem in the form of \cite[Theorem 3.2]{Bellamy19} applies, since $\Pi$ has integral vertices by the saturation of Proposition~\ref{sh-0007}.
\end{proof}
