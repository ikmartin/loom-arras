\begin{corollary}[Trees]\label{sh-000A}
A quiver whose underlying graph is a forest has $\Zcyc(\quiv) = 0$.
\end{corollary}
\begin{proof}
\uses{sh-0009}
A forest on $|V(\quiv)|$ vertices with $c$ components has exactly $|V(\quiv)| - c$ edges, so Theorem~\ref{sh-0009} and \eqref{eq:rank} give rank zero, and a saturated sublattice of rank zero is trivial.
\end{proof}
