\begin{definition}[Balanced weighting]\label{sh-0004}
A weighting $w$ is \emph{balanced} if $\bd w = 0$, and the balanced weightings form the sublattice
\begin{equation}\label{eq:cycle-lattice}
\Zcyc(\quiv) \;=\; \ker \bd \;\subseteq\; W(\quiv),
\end{equation}
which we call the \emph{cycle lattice} of $\quiv$.
\end{definition}
