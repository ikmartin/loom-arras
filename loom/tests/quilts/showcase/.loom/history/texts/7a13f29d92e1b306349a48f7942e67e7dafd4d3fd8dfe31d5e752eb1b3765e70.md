\begin{definition}[Divergence]\label{sh-0003}
The \emph{divergence} of a weighting $w$ at a vertex $v$ is
\begin{equation}\label{eq:divergence}
(\bd w)(v) \;=\; \sum_{a \colon t(a) = v} w(a) \;-\; \sum_{a \colon s(a) = v} w(a),
\end{equation}
the weight arriving at $v$ less the weight leaving it. The resulting homomorphism $\bd \colon W(\quiv) \to \ZZ^{V}$ is the \emph{divergence map}.
\end{definition}
