\begin{lemma}[Orbits]\label{sy-0002}
Every orbit of a widget has one or two points; compare \eqref{eq:fix} in Definition~\ref{sy-0001}.
\end{lemma}
\begin{proof}
The orbit of $x$ is $\{x, \sigma x\}$ because
\begin{equation}\label{eq:step}
\sigma(\sigma x) = x
\end{equation}
by \eqref{eq:prose}.
\end{proof}
