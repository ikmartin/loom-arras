\section{Introduction}\label{sy-0100}

This paper studies widgets. The basic identity is
\begin{equation}\label{eq:prose}
\sigma^2 = \mathrm{id},
\end{equation}
which we use throughout.\footnote{The name comes from nowhere.} The \GW{} theory of gadgets is not touched. See \cite[Chapter II]{Har77} for schemes and \cite[Theorem 2.1]{Kre99} for cycle groups; the lemma \cite[Lemma 9]{Kre99} is not in the digest. Note also the dangling reference to Lemma~\ref{sy-9999}.

\begin{definition}[Widget]\label{sy-0001}
A \emph{widget} is a pair $(X,\sigma)$ of a set and an involution $\sigma$ of $X$, so that \eqref{eq:prose} holds. Its fixed locus is
\begin{equation}\label{eq:fix}
\Fix(\sigma) = \{x \in X : \sigma x = x\}.
\end{equation}
\end{definition}

% !LOOM author: The synthetic quilt
% !LOOM created: 2026-09-16
% !LOOM tags: orbits, setup

\begin{lemma}[Orbits]\label{sy-0002}
Every orbit of a widget has one or two points; compare \eqref{eq:fix} in Definition~\ref{sy-0001}.
\end{lemma}
\begin{proof}
The orbit of $x$ is $\{x, \sigma x\}$ because
\begin{equation}\label{eq:step}
\sigma(\sigma x) = x
\end{equation}
by \eqref{eq:prose}.
\end{proof}

\begin{theorem}[Parity]\label{sy-0003}
Let $(X,\sigma)$ be a finite widget. Then $|X| \equiv |\Fix(\sigma)| \pmod 2$.
\end{theorem}
\begin{proof}[First proof]\label{sy-0004}
\uses{sy-0002}
By Lemma~\ref{sy-0002} the set $X$ is a disjoint union of orbits of size one or two.
\end{proof}
\begin{proof}[Second proof]\label{sy-0005}
\uses{sy-0002, sy-0001}
Count fixed points of the permutation $\sigma$ directly, using \eqref{eq:step} from the proof of Lemma~\ref{sy-0002} and Definition~\ref{sy-0001}.
\end{proof}
