\begin{theorem}[{\cite[Theorem 3.2, p.~2]{Bellamy19}}]\label{Bellamy19-thm-3.2}
Let $D$ be a weighted digraph whose weight function is balanced at every vertex. Then the counting function of the median polytope of $D$ agrees with a polynomial $L_{D}$ of degree $\dim M(D)$ for every $k \geq 0$, and the leading coefficient of $L_{D}$ is the relative volume of $M(D)$.
\end{theorem}
